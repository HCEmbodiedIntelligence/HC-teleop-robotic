cmake_minimum_required(VERSION 3.16)
project(hc_kinematics VERSION 1.0 LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(hc_kinematics SHARED src/kinematics.cpp)
target_link_libraries(hc_kinematics PRIVATE Eigen3::Eigen)
target_compile_features(hc_kinematics PRIVATE cxx_std_17)
target_compile_options(hc_kinematics PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(hc_kinematics PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
install(TARGETS hc_kinematics LIBRARY DESTINATION lib)
