cmake_minimum_required(VERSION 3.16)
project(hc_control VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HC_CONTROL_SANITIZERS "Enable ASan and UBSan in development" OFF)
find_package(Threads REQUIRED)
add_library(hc_control_core src/core.cpp src/storage.cpp)
set_source_files_properties(src/storage.cpp PROPERTIES COMPILE_OPTIONS "-O0")
target_include_directories(hc_control_core PUBLIC include)
target_compile_options(hc_control_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(hc_control_core PUBLIC Threads::Threads)
add_executable(hc-control src/main.cpp)
target_include_directories(hc-control PRIVATE vendor)
target_link_libraries(hc-control PRIVATE hc_control_core)
target_compile_options(hc-control PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -O1)
if(HC_CONTROL_SANITIZERS)
  foreach(target hc_control_core hc-control)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(hc-control-tests tests/core_test.cpp)
  target_link_libraries(hc-control-tests PRIVATE hc_control_core)
  # Large fixed-capacity fixtures do not benefit from costly O2 inlining. The
  # production control library retains the selected build-type optimization.
  target_compile_options(hc-control-tests PRIVATE -O0)
  add_test(NAME control_contracts COMMAND hc-control-tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME control_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ipc_test.py $<TARGET_FILE:hc-control>)
  endif()
endif()
install(TARGETS hc-control RUNTIME DESTINATION bin)
